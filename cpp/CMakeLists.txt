cmake_minimum_required(VERSION 3.16)
project(ExecCore LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_compile_options(-Wall -Wextra -Wpedantic)
include_directories(include)
add_library(execcore STATIC
  src/order.cpp src/execution_engine.cpp src/exchange_simulator.cpp src/order_book.cpp)
target_link_libraries(execcore PUBLIC pthread)
add_executable(execcore_demo src/main.cpp)
target_link_libraries(execcore_demo PRIVATE execcore)
add_executable(execcore_tests tests/test_main.cpp)
target_link_libraries(execcore_tests PRIVATE execcore)
add_executable(orderbook_bench tests/orderbook_bench.cpp)
target_link_libraries(orderbook_bench PRIVATE execcore)
enable_testing()
add_test(NAME execcore_tests COMMAND execcore_tests)
